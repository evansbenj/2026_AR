# Simulations

I'm interested in evaluating how frequently and how quickly turnover events happen with dominant cis, dominant trans, and recessive trans. Here are simulations. All start with a WZ/ZZ system.

#!/usr/bin/env perl
use strict;
use warnings;
#use lib qw(~/perl_modules);
use List::MoreUtils qw/ uniq /;

# This script will run a simulation that explores how long it takes
# for a neoW dominant mutation to fix and cause a cis sex chromosome turnover
# if genotypes change over generations of random mating

# It will start with 50:50 ratio of WZAA females and ZZAA males
# a neoW mutation will appear in either sex and convert this individual to a female

# to execute:
# ./New_dominantW_sims.pl outputfilename

my $outputfile = $ARGV[0];

unless (open(OUTFILE, ">$outputfile"))  {
	print "I can\'t write to $outputfile\n";
	exit;
}
print "Creating output file: $outputfile\n";

print OUTFILE "Generation	Resolution	SexRatio\n";

my $population_size=1000; # use an even number
my $a;
my $x;
my $y;
my @mothers = ();
my @fathers = ();
my $mother;
my $father;
my @daughters = ();
my @sons = ();
my @genotype_countz = ();
my @sex_chr_countz = ();
my $num;
my $number_of_generations=0;
my $resolution=0;
my @unique_mothers;
my @unique_fathers;


for ($a = 0 ; $a < 1000000 ; $a++ ) { # do 1 million simulations
	$resolution = 0;
	$number_of_generations=0;
	@mothers = ();
	@fathers = ();
	# female genotypes: 0 = WZAA, 1 = WZWA, 2 = ZZWA
	# male genotypes: 3 = ZZAA
	
	# first populate the parent array with an equal frequency of each genotype
	
	# mothers
	for ($y = 0 ; $y < $population_size/2 ; $y++ ) {
		push(@mothers, 0); # 0 = WZAA
	}	
	# fathers
	for ($y = 0 ; $y < $population_size/2 ; $y++ ) {
		push(@fathers, 3); # 3 = ZZAA
	}
	if (@mothers == 0 || @fathers == 0) {
	    $resolution = 3;
	    last;
	}

	# now run the simulation until a two chromosome system fixes
	until($resolution != 0) {
		# first generate offspring genotypes
		# allow a sex ratio skew
		@daughters = ();
		@sons = ();
		for ($y = 0 ; $y < $population_size ; $y++ ) {
			# choose parents randomly with replacement - approximates a Poisson distribution of reproductive variation
			$mother = $mothers[rand @mothers];
			$father = $fathers[rand @fathers];		
			if($mother == 0){ # WZAA x ZZAA the child is either a WZAA daughter or a ZZAA son; fathers are always ZZAA
				# flip a coin
				$x = int(rand(2)); # this should generate a random 0 or 1
				if($x == 0){
					push(@daughters,0); # WZAA daughter
					# introduce a mutation if this is the first child
					if(($number_of_generations == 0)&&($y ==0)){
						# this is the first offspring of the first generation; give her a neoW
						# female genotypes: 0 = WZAA, 1 = WZWA, 2 = ZZWA
						# male genotypes: 3 = ZZAA
						# possible new genotypes: 1 = WZXA (daughter) 
						$daughters[0] = 1; # WZXA daughter	
					}
				}
				elsif($x == 1){
					push(@sons,3); # ZZAA son
					# introduce a mutation if this is the first child
					if(($number_of_generations == 0)&&($y ==0)){
						# this is the first offspring of the first generation; give her a neoW
						# female genotypes: 0 = WZAA, 1 = WZWA, 2 = ZZWA
						# male genotypes: 3 = ZZAA
						# possible new genotypes: 1 = WZXA (daughter) 
						pop @sons; # remove this son because they are now a female
						push(@daughters,2); # ZZWA daughter	
					}
				}
			}
			elsif($mother == 1){ # WZWA x ZZAA the child can be: WZWA or WZXA daughter ; fathers are always ZZAA
				# flip a coin
				$x = int(rand(4)); # this should generate a random 0 or 1 or 2 or 3
				if($x == 0){ 
					push(@daughters,0); # WZAA daughter
				}
				elsif($x == 1){
					push(@daughters,1); # WZWA daughter
				}
				elsif($x == 2){
					push(@daughters,2); # ZZWA daughter
				}
				elsif($x == 3){
					push(@sons,3); # ZZAA son
				}
			}
			elsif($mother == 2){ # ZZWA x ZZAA the child can be: ZZWA daughter or ZZAA son
				# flip a coin
				$x = int(rand(2)); # this should generate a random 0 or 1 or 2 or 3
				if($x == 0){ 
					push(@daughters,2); # ZZWA daughter
				}
				elsif($x == 1){
					push(@sons,3); # ZZAA daughter
				}	
			}
		}
		# offspring are generated, now assign them to be parents of the next generation
		@mothers = @daughters;
		@fathers = @sons;
		# female genotypes: 0 = WZAA, 1 = WZWA, 2 = ZZWA
		# male genotypes: 3 = ZZAA
		# resolution_1: 0 = WZAA, 3 = ZZAA (no change in heterogamy)
		# resolution_2: 2 = ZZWA, 3 = ZZXA (change in heterogamy)
		# check if we have resolution
		@unique_mothers = uniq @mothers;
		@unique_fathers = uniq @fathers;
		my $sexratio = "NA";
		$sexratio = (@fathers / @mothers) if @mothers && @fathers;
		
		if((@unique_mothers  == 1)&&($unique_mothers[0] == "0"))
			{
			   $resolution = 1;
			   # print outfile
			   print OUTFILE $number_of_generations,"\t",$resolution,"\t",$sexratio,"\n";
		}
		elsif((@unique_mothers  == 1)&&($unique_mothers[0] == "2"))
		   {
			 $resolution = 2;
			print OUTFILE $number_of_generations,"\t",$resolution,"\t",$sexratio,"\n";
	 		print "res2 $number_of_generations and @unique_mothers and @unique_fathers\n";
		}
		elsif((@mothers == 0)||(@fathers == 0)){
			# extinction due to sex ratio
				$resolution = 3;
			print OUTFILE $number_of_generations,"\t",$resolution,"\t",$sexratio,"\n";
	 		print "res3 $number_of_generations and @unique_mothers and @unique_fathers\n";			
		}
	$number_of_generations+=1;
	}
}

close OUTFILE;

			   print OUTFILE $number_of_generations,"\t",$resolution,"\t",$sexratio,"\n";
		}
		elsif(((@unique_mothers  == 1)&&($unique_mothers[0] == "3"))
		   &&
		   ((@unique_fathers  == 1)&&($unique_fathers[0] == "5"))
		   ){
			 $resolution = 2;
			print OUTFILE $number_of_generations,"\t",$resolution,"\t",$sexratio,"\n";
	 		print "res2 $number_of_generations and @unique_mothers and @unique_fathers\n";
		}
		elsif((@mothers == 0)||(@fathers == 0)){
			# extinction due to sex ratio
				$resolution = 3;
			print OUTFILE $number_of_generations,"\t",$resolution,"\t",$sexratio,"\n";
	 		print "res3 $number_of_generations and @unique_mothers and @unique_fathers\n";			
		}
	$number_of_generations+=1;
	}
}

close OUTFILE;
